
1. Windows PowerShell – project and folder commands
# Navigate to project
cd "YOUR_PROJECT_FOLDER"

# Show current folder
Get-Location

# List files and folders
Get-ChildItem

# List all project files
Get-ChildItem -Recurse -File

# Find a particular file
Get-ChildItem -Recurse -Filter "*.py"

# Find text in project files
findstr /S /I "operational" *.*

# Check Python version
python --version

# Check pip version
pip --version

# Check Git version
git --version

# Check Docker version
docker --version

# Check Docker Compose version
docker compose version

3. Docker commands
# List running containers
docker ps

# List all containers, including stopped
docker ps -a

# Check Docker Compose projects
docker compose ls

# List Docker volumes
docker volume ls

# List Docker networks
docker network ls

# Start project containers
docker compose up -d

# Start and rebuild images
docker compose up -d --build

# Stop project containers
docker compose stop

# Start stopped project containers
docker compose start

# Restart all services
docker compose restart

# Restart specific services
docker compose restart airflow-scheduler airflow-webserver airflow-worker

# Check container logs
docker compose logs --tail 100

# Follow logs
docker compose logs -f

# Stop and remove project containers
docker compose down

# Check container details
docker inspect CONTAINER_NAME

# Check Docker disk usage
docker system df

7. Airflow connection commands
# List all connections
docker exec -it airflow-docker-airflow-scheduler-1 airflow connections list

# Inspect PostgreSQL connection
docker exec -it airflow-docker-airflow-scheduler-1 airflow connections get postgres_insurance

# Create PostgreSQL connection
docker exec -it airflow-docker-airflow-scheduler-1 airflow connections add postgres_insurance --conn-type postgres --conn-host postgres --conn-login postgres --conn-password YOUR_PASSWORD --conn-port 5432 --conn-schema postgres

# Delete a connection (only if you really intend to recreate it)
docker exec -it airflow-docker-airflow-scheduler-1 airflow connections delete postgres_insurance

# Check whether Docker can resolve PostgreSQL hostname
docker exec -it airflow-docker-airflow-scheduler-1 python -c "import socket; print(socket.gethostbyname('postgres'))"
